/**
 * 淇濆瓨浜哄伐淇銆? * id: ORN-TXT-* | ORN-IMG-*
 * patch:
 *  - text: { classify?, entities?, relations?, structured? }
 *  - vision: { ocr?, yolo? }
 */
export async function saveCorrection(id, patch = {}) {
  await businessDelay()
  const auth = useAuthStore()

  const text = state.texts.find((x) => x.id === id)
  if (text) {
    const before = text.corrected ? clone(text.corrected) : clone(text.model)
    const next = {
      classify: patch.classify ? clone(patch.classify) : clone(before.classify),
      entities: patch.entities ? clone(patch.entities) : clone(before.entities),
      relations: patch.relations ? clone(patch.relations) : clone(before.relations),
      structured: patch.structured ? clone(patch.structured) : clone(before.structured),
    }
    const changed = diffTextFields(before, next)
    text.corrected = next
    text.status = 'corrected'
    text.changeLog.unshift({
      at: now(),
      actor: actorName(),
      fields: changed,
    })
    pushAudit({
      action: '淇濆瓨鏂囨湰浜哄伐淇',
      detail: `${id} 路 鍙樻洿 ${changed.join(',') || '鏃?}`,
    })
    return { id, kind: 'text', corrected: clone(next), changedFields: changed }
  }

  const vision = state.visions.find((x) => x.id === id)
  if (vision) {
    const changed = []
    if (patch.ocr) {
      vision.ocr = clone(patch.ocr)
      vision.ocrCorrected = true
      changed.push('ocr')
    }
    if (patch.yolo) {
      vision.yolo = clone(patch.yolo)
      vision.yoloCorrected = true
      changed.push('yolo')
    }
    vision.status = 'corrected'
    vision.changeLog.unshift({ at: now(), actor: actorName(), fields: changed })
    pushAudit({
      action: '淇濆瓨瑙嗚浜哄伐淇',
      detail: `${id} 路 鍙樻洿 ${changed.join(',') || '鏃?}`,
    })
    return {
      id,
      kind: 'vision',
      ocr: clone(vision.ocr),
      yolo: clone(vision.yolo),
      changedFields: changed,
    }
  }

  throw new Error(`棰勬爣瀵硅薄涓嶅瓨鍦細${id}`)
}

function diffTextFields(before, after) {
  const fields = []
  if (JSON.stringify(before.classify) !== JSON.stringify(after.classify)) fields.push('classify')
  if (JSON.stringify(before.entities) !== JSON.stringify(after.entities)) fields.push('entities')
  if (JSON.stringify(before.relations) !== JSON.stringify(after.relations)) fields.push('relations')
  if (JSON.stringify(before.structured) !== JSON.stringify(after.structured)) fields.push('structured')
  return fields
}

/**
 * 鎺ㄨ繘楠屾敹鍚戝涓€姝ャ€? * steps: confirm 鈫?annotate 鈫?qa 鈫?accept
 * 鏈€缁?accept 浠?reviewer/admin 鍙墽琛屻€? */
export async function advanceAcceptStep(id, { qaNote } = {}) {
  await businessDelay()
  const auth = useAuthStore()
  const j = state.accepts.find((x) => x.id === id)
  if (!j) throw new Error(`楠屾敹浠诲姟涓嶅瓨鍦細${id}`)
  if (j.status === 'accepted') throw new Error('浠诲姟宸查獙鏀跺畬鎴?)

  const order = ACCEPT_STEPS.map((s) => s.key)
  const idx = order.indexOf(j.step)
  if (idx < 0) throw new Error(`鏈煡姝ラ锛?{j.step}`)

  if (j.step === 'qa' && qaNote != null) j.qaNote = qaNote

  // completing current step
  if (!j.stepDone.includes(j.step)) j.stepDone.push(j.step)

  const isLast = idx === order.length - 1
  if (isLast) {
    if (!['reviewer', 'admin'].includes(auth.role)) {
      throw new Error('浠呭鏍稿憳鎴栫鐞嗗憳鍙渶缁堥獙鏀?)
    }
    j.status = 'accepted'
    j.acceptedAt = now()
    j.acceptedBy = actorName()
    pushAudit({ action: '棰勬爣缁撴灉楠屾敹閫氳繃', detail: `${id} 路 ${j.title}` })
    return { ...j, stepDone: [...j.stepDone] }
  }

  j.step = order[idx + 1]
  pushAudit({
    action: `楠屾敹鎺ㄨ繘锛?{ACCEPT_STEPS[idx].label}`,
    detail: `${id} 鈫?${ACCEPT_STEPS[idx + 1].label}`,
  })
  return { ...j, stepDone: [...j.stepDone] }
}

/**
 * 淇濆瓨鍔ㄤ綔璧锋甯ф爣娉ㄣ€? * @param {{ taskId: string, startFrame: number, endFrame: number, action: string, instruction: string, keyNode?: boolean, id?: string }} payload
 */
export async function saveActionSpan(payload) {
  await businessDelay()
  const t = state.tasks.find((x) => x.id === payload.taskId)
  if (!t) throw new Error(`閲囬泦浠诲姟涓嶅瓨鍦細${payload.taskId}`)

  const startFrame = Number(payload.startFrame)
  const endFrame = Number(payload.endFrame)
  if (!Number.isFinite(startFrame) || !Number.isFinite(endFrame)) {
    throw new Error('璧锋甯у繀椤讳负鏁板瓧')
  }
  if (startFrame < 0 || endFrame > t.totalFrames) {
    throw new Error(`甯ц寖鍥撮渶鍦?0鈥?{t.totalFrames}`)
  }
  if (endFrame <= startFrame) throw new Error('缁撴潫甯у繀椤诲ぇ浜庤捣濮嬪抚')
  if (!payload.action) throw new Error('璇峰～鍐欏姩浣滅紪鐮?)

  const tpl = ACTION_TEMPLATES.find((a) => a.code === payload.action)
  const actionLabel = tpl?.name || payload.action

  if (payload.id) {
    const row = t.actions.find((a) => a.id === payload.id)
    if (!row) throw new Error(`鍔ㄤ綔鐗囨涓嶅瓨鍦細${payload.id}`)
    Object.assign(row, {
      startFrame,
      endFrame,
      action: payload.action,
      actionLabel,
      instruction: payload.instruction || '',
      keyNode: !!payload.keyNode,
    })
  } else {
    t.actions.push({
      id: `act-${Date.now()}-${t.actions.length + 1}`,
      startFrame,
      endFrame,
      action: payload.action,
      actionLabel,
      instruction: payload.instruction || '',
      keyNode: !!payload.keyNode,
    })
  }

  t.actions.sort((a, b) => a.startFrame - b.startFrame)
  t.status = 'annotating'
  pushAudit({
    action: '淇濆瓨鍔ㄤ綔璧锋甯?,
    detail: `${t.id} 路 ${actionLabel} 路 F${startFrame}-F${endFrame}`,
  })
  return clone(t)
}

/** 鏍囪涓嶅悎鏍兼牱鏈?*/
export async function markUnqualified(taskId, issueIds) {
  await businessDelay()
  const t = state.tasks.find((x) => x.id === taskId)
  if (!t) throw new Error(`閲囬泦浠诲姟涓嶅瓨鍦細${taskId}`)
  const set = new Set(issueIds || [])
  let count = 0
  for (const i of t.issues) {
    if (set.has(i.id)) {
      i.unqualified = true
      count += 1
    }
  }
  pushAudit({
    action: '鏍囪涓嶅悎鏍兼牱鏈?,
    detail: `${taskId} 路 ${count} 鏉,
  })
  return clone(t.issues)
}

    visible: clone(t.issues.filter((i) => !i.filtered)),

  }
}

  const personnel = task.personnelIds.map((id) => {
    const u = users.find((x) => x.id === id)
    return u ? `${u.name}(${u.role})` : id
  })

  return {
    meta: {
      taskId: task.id,
      name: task.name,
      scene: sceneLabel,
      plan: planLabel,
      mode: modeLabel,
      instruction: task.instruction,
      fps: task.fps,
      totalFrames: task.totalFrames,
      personnel,
      exportedAt: now(),
      exportedBy: actorName(),
      product: 'Orion Annotate',
    },
    actions: task.actions.map((a) => ({
      id: a.id,
      startFrame: a.startFrame,
      endFrame: a.endFrame,
      action: a.action,
      actionLabel: a.actionLabel,
      instruction: a.instruction,
      keyNode: a.keyNode,
    })),
    issues: task.issues.map((i) => ({
      id: i.id,
      type: i.type,
      frame: i.frame,
      severity: i.severity,
      message: i.message,
      sampleId: i.sampleId,
      unqualified: !!i.unqualified,
    })),
  }
}

function buildExportRows(task) {
  if (task.actions.length === 0) {
    return [
      {
        taskId: task.id,
        taskName: task.name,
        startFrame: '',
        endFrame: '',
        action: '',
        actionLabel: '',
        instruction: task.instruction,
        keyNode: '',
      },
    ]
  }
  return task.actions.map((a) => ({
    taskId: task.id,
    taskName: task.name,
    startFrame: a.startFrame,
    endFrame: a.endFrame,
    action: a.action,
    actionLabel: a.actionLabel,
    instruction: a.instruction,
    keyNode: a.keyNode ? 'Y' : 'N',
  }))
}

/**
 * 瀵煎嚭 JSON / Excel / CSV锛屾枃浠跺悕 ORN-VLA-{taskId}-{timestamp}.{ext}
 * @param {string} taskId
 * @param {'json'|'excel'|'csv'} format
 */
export async function exportVla(taskId, format) {
  await businessDelay()
  const t = state.tasks.find((x) => x.id === taskId)
  if (!t) throw new Error(`閲囬泦浠诲姟涓嶅瓨鍦細${taskId}`)

  const ts = dayjs().format('YYYYMMDDHHmmss')
  // Prefer ORN-VLA-{taskId}-{ts}; if taskId already starts with ORN-VLA-, do not double-prefix
  const base = taskId.startsWith('ORN-VLA-') ? `${taskId}-${ts}` : `ORN-VLA-${taskId}-${ts}`
  const payload = buildExportPayload(t)
  const rows = buildExportRows(t)

  if (format === 'json') {
    const filename = `${base}.json`
    exportJSON(filename, payload)
    pushAudit({ action: '瀵煎嚭 VLA JSON', detail: filename })
    return { filename, format }
  }
  if (format === 'csv') {
    const filename = `${base}.csv`
    exportCSV(filename, rows)
    pushAudit({ action: '瀵煎嚭 VLA CSV', detail: filename })
    return { filename, format }
  }
  if (format === 'excel') {
    const filename = `${base}.xlsx`
    exportExcel(filename, rows)
    pushAudit({ action: '瀵煎嚭 VLA Excel', detail: filename })
    return { filename, format }
  }
  throw new Error(`涓嶆敮鎸佺殑瀵煎嚭鏍煎紡锛?{format}`)
}

  const base = taskId.startsWith('ORN-VLA-') ? `${taskId}-${timestamp}` : `ORN-VLA-${taskId}-${timestamp}`
  return `${base}.${ext}`
}

/**
 * 鍒涘缓璁粌瀹為獙锛堜粎绠＄悊鍛橈級
 * @param {{ name: string, algorithm: string, version: string, model: string, batchSize: number, learningRate: number, epochs: number, augment: boolean, augmentOps?: string[], gpuIds?: string[] }} cfg
 */
export async function createExperiment(cfg) {
  await businessDelay()
  requireAdmin('鍒涘缓瀹為獙')
  if (!cfg?.name?.trim()) throw new Error('璇峰～鍐欏疄楠屽悕绉?)
  if (!cfg.algorithm) throw new Error('璇烽€夋嫨绠楁硶')
  if (!cfg.version) throw new Error('璇烽€夋嫨绠楁硶鐗堟湰')
  if (!cfg.model) throw new Error('璇烽€夋嫨妯″瀷')
  const epochs = Number(cfg.epochs)
  const batchSize = Number(cfg.batchSize)
  const learningRate = Number(cfg.learningRate)
  if (!Number.isFinite(epochs) || epochs < 1) throw new Error('epoch 闇€涓烘鏁存暟')
  if (!Number.isFinite(batchSize) || batchSize < 1) throw new Error('batchSize 鏃犳晥')
  if (!Number.isFinite(learningRate) || learningRate <= 0) throw new Error('learningRate 鏃犳晥')

  const id = nextExpId(state.seq++)
  const gpuIds = Array.isArray(cfg.gpuIds) ? [...cfg.gpuIds] : []
  const exp = {
    id,
    name: cfg.name.trim(),
    algorithm: cfg.algorithm,
    version: cfg.version,
    model: cfg.model,
    batchSize,
    learningRate,
    epochs,
    augment: !!cfg.augment,
    augmentOps: cfg.augment ? [...(cfg.augmentOps || [])] : [],
    gpuIds,
    status: gpuIds.length ? 'ready' : 'draft',
    currentEpoch: 0,
    createdAt: now(),
    startedAt: null,
    finishedAt: null,
    logs: [{ at: now(), level: 'info', message: '瀹為獙宸插垱寤? }],
    metrics: { loss: [], acc: [], map: [] },
  }

  if (gpuIds.length) {
    for (const gid of gpuIds) {
      const card = state.gpus.find((g) => g.id === gid)
      if (!card) throw new Error(`GPU 涓嶅瓨鍦細${gid}`)
      if (!card.free && !EXPERIMENTS.some((e) => e.gpuIds.includes(gid))) {
        // allow if currently free in runtime state
      }
      if (!card.free) throw new Error(`GPU 宸茶鍗犵敤锛?{card.name}`)
      card.free = false
    }
    const names = gpuIds.map((id) => state.gpus.find((g) => g.id === id)?.name || id).join(', ')
    exp.logs.push({ at: now(), level: 'info', message: `鍒涘缓鏃跺垎閰?GPU锛?{names}` })
  }

  state.experiments.unshift(exp)
  pushAudit({
    action: '鍒涘缓璁粌瀹為獙',
    detail: `${id} 路 ${exp.name} 路 ${exp.algorithm}@${exp.version} 路 ${exp.model}`,
  })
  return clone(exp)
}

/**
 * 鍒嗛厤 GPU锛堜粎绠＄悊鍛橈級
 * @param {string} expId
 * @param {string[]} gpuIds
 */
export async function allocateGpu(expId, gpuIds) {
  await businessDelay()
  requireAdmin('鍒嗛厤 GPU')
  const exp = findExp(expId)
  if (exp.status === 'running') throw new Error('杩愯涓殑瀹為獙涓嶅彲鏀归厤 GPU')
  if (!gpuIds?.length) throw new Error('璇疯嚦灏戦€夋嫨涓€寮?GPU')

  // release previous
  for (const gid of exp.gpuIds) {
    const card = state.gpus.find((g) => g.id === gid)
    if (card) card.free = true
  }

  for (const gid of gpuIds) {
    const card = state.gpus.find((g) => g.id === gid)
    if (!card) throw new Error(`GPU 涓嶅瓨鍦細${gid}`)
    // allow re-selecting same cards already on this exp
    const heldByOther = state.experiments.some(
      (e) => e.id !== expId && e.gpuIds.includes(gid) && e.status !== 'finished',
    )
    if (!card.free && heldByOther) throw new Error(`GPU 宸茶鍗犵敤锛?{card.name}`)
    card.free = false
  }

  exp.gpuIds = [...gpuIds]
  if (exp.status === 'draft') exp.status = 'ready'
  const names = gpuIds.map((id) => state.gpus.find((g) => g.id === id)?.name || id).join(', ')
  exp.logs.push({ at: now(), level: 'info', message: `鍒嗛厤 GPU锛?{names}` })
  pushAudit({ action: '鍒嗛厤璁粌 GPU', detail: `${expId} 路 ${names}` })
  return clone(exp)
}

/** 鍚姩璁粌锛堜粎绠＄悊鍛橈級 */
export async function startTrain(expId) {
  await businessDelay()
  requireAdmin('鍚姩璁粌')
  const exp = findExp(expId)
  if (!exp.gpuIds.length) throw new Error('璇峰厛鍒嗛厤 GPU')
  if (exp.status === 'running') throw new Error('瀹為獙宸插湪杩愯')
  if (exp.status === 'finished' && exp.currentEpoch >= exp.epochs) {
    // allow restart: reset metrics
    exp.metrics = { loss: [], acc: [], map: [] }
    exp.currentEpoch = 0
    exp.finishedAt = null
  }
  exp.status = 'running'
  exp.startedAt = now()
  exp.logs.push({
    at: now(),
    level: 'info',
    message: `寮€濮嬭缁?路 epoch 鐩爣 ${exp.epochs} 路 澧炲己 ${exp.augment ? '寮€' : '鍏?}`,
  })
  pushAudit({ action: '鍚姩璁粌', detail: `${expId} 路 epochs=${exp.epochs}` })
  return clone(exp)
}

/** 鍋滄璁粌锛堜粎绠＄悊鍛橈級 */
export async function stopTrain(expId) {
  await businessDelay()
  requireAdmin('鍋滄璁粌')
  const exp = findExp(expId)
  if (exp.status !== 'running') throw new Error('褰撳墠涓嶅湪杩愯鐘舵€?)
  exp.status = exp.currentEpoch >= exp.epochs ? 'finished' : 'stopped'
  if (exp.status === 'finished') exp.finishedAt = now()
  exp.logs.push({
    at: now(),
    level: 'warning',
    message: `鎵嬪姩鍋滄 路 宸插畬鎴?epoch ${exp.currentEpoch}/${exp.epochs}`,
  })
  pushAudit({ action: '鍋滄璁粌', detail: `${expId} 路 epoch ${exp.currentEpoch}` })
  return clone(exp)
}

/**
 * 鎺ㄨ繘涓€涓?mock epoch锛堢敱 TrainRunner 瀹氭椂鍣ㄨ皟鐢級
 * @returns {{ experiment: object, point: {loss,acc,map}, done: boolean }}
 */
export async function tickEpoch(expId) {
  // short delay so UI stays snappy during timer ticks
  await businessDelay()
  const exp = findExp(expId)
  if (exp.status !== 'running') {
    return { experiment: clone(exp), point: null, done: exp.status === 'finished' }
  }
  if (exp.currentEpoch >= exp.epochs) {
    exp.status = 'finished'
    exp.finishedAt = now()
    return { experiment: clone(exp), point: null, done: true }
  }

  const next = exp.currentEpoch + 1
  const point = mockMetricPoint(next, exp.epochs)
  exp.currentEpoch = next
  exp.metrics.loss.push(point.loss)
  exp.metrics.acc.push(point.acc)
  exp.metrics.map.push(point.map)
  exp.logs.push({
    at: now(),
    level: 'info',
    message: `epoch ${next}/${exp.epochs} 路 loss=${point.loss} acc=${point.acc} mAP=${point.map}`,
  })

  let done = false
  if (next >= exp.epochs) {
    exp.status = 'finished'
    exp.finishedAt = now()
    exp.logs.push({ at: now(), level: 'success', message: `璁粌瀹屾垚 epoch ${next}/${exp.epochs}` })
    done = true
    pushAudit({ action: '璁粌瀹屾垚', detail: `${expId} 路 ${next} epochs` })
  }
  return { experiment: clone(exp), point, done }
}

/**
 * 瀵煎嚭璁粌鎶ヨ〃 CSV / Excel
 * @param {string} expId
 * @param {'csv'|'excel'} format
 */
export async function exportReport(expId, format) {
  await businessDelay()
  const exp = findExp(expId)
  const rows = []
  const n = Math.max(exp.metrics.loss.length, exp.metrics.acc.length, exp.metrics.map.length)
  for (let i = 0; i < n; i += 1) {
    rows.push({
      experimentId: exp.id,
      name: exp.name,
      epoch: i + 1,
      loss: exp.metrics.loss[i] ?? '',
      acc: exp.metrics.acc[i] ?? '',
      map: exp.metrics.map[i] ?? '',
      algorithm: exp.algorithm,
      version: exp.version,
      model: exp.model,
      augment: exp.augment ? 'Y' : 'N',
      status: exp.status,
    })
  }
  if (rows.length === 0) {
    rows.push({
      experimentId: exp.id,
      name: exp.name,
      epoch: '',
      loss: '',
      acc: '',
      map: '',
      algorithm: exp.algorithm,
      version: exp.version,
      model: exp.model,
      augment: exp.augment ? 'Y' : 'N',
      status: exp.status,
    })
  }

  const ts = dayjs().format('YYYYMMDDHHmmss')
  const base = `${exp.id}-report-${ts}`
  if (format === 'csv') {
    const filename = `${base}.csv`
    exportCSV(filename, rows)
    pushAudit({ action: '瀵煎嚭璁粌鎶ヨ〃 CSV', detail: filename })
    return { filename, format }
  }
  if (format === 'excel') {
    const filename = `${base}.xlsx`
    exportExcel(filename, rows)
    pushAudit({ action: '瀵煎嚭璁粌鎶ヨ〃 Excel', detail: filename })
    return { filename, format }
  }
  throw new Error(`涓嶆敮鎸佺殑瀵煎嚭鏍煎紡锛?{format}`)
}
