<script setup>import { notification } from 'ant-design-vue'

import { computed, onMounted, reactive, ref } from 'vue'
import { useRouter } from 'vue-router'
import {
  listQcTasks,
  createQcTask,
  assignTask,
  listMeta,
  getModalityMeta,
  getStatusMeta,
} from '@/services/qc'
import { useAuthStore } from '@/stores/auth'

const router = useRouter()
const auth = useAuthStore()

const loading = ref(false)
const tasks = ref([])
const meta = reactive({ modalities: [], statuses: [], priorities: [], users: [] })
const filters = reactive({ keyword: '', status: '', modality: '' })

const createVisible = ref(false)
const createForm = reactive({
  title: '',
  modality: 'text_ner',
  priority: 'medium',
  sampleCount: 20,
  batchId: '',
})

const assignVisible = ref(false)
const assignTarget = ref(null)
const assignUserId = ref('')

const annotators = computed(() =>
  meta.users.filter((u) => u.role === 'annotator' || u.role === 'admin'),
)

async function load() {
  loading.value = true
  try {
    Object.assign(meta, listMeta())
    tasks.value = await listQcTasks({ ...filters })
  } finally {
    loading.value = false
  }
}

onMounted(load)

async function onSearch() {
  await load()
}

async function onCreate() {
  if (!createForm.title.trim()) {
    notification.warning({ message: '璇峰～鍐欏伐鍗曟爣棰?, placement: 'topRight' })
    return
  }
  const t = await createQcTask({ ...createForm })
  notification.success({ message: `宸插垱寤哄伐鍗?${t.id}`, placement: 'topRight' })
  createVisible.value = false
  createForm.title = ''
  await load()
}

function openAssign(row) {
  assignTarget.value = row
  assignUserId.value = row.assigneeId || ''
  assignVisible.value = true
}

async function confirmAssign() {
  if (!assignUserId.value) {
    notification.warning({ message: '璇烽€夋嫨澶勭悊浜?, placement: 'topRight' })
    return
  }
  await assignTask(assignTarget.value.id, assignUserId.value)
  notification.success({ message: '鍒嗛厤鎴愬姛', placement: 'topRight' })
  assignVisible.value = false
  await load()
}

function goAnnotate(id) {
  router.push(`/qc/annotate/${id}`)
}

function goAudit(id) {
  router.push(`/qc/audit/${id}`)
}

function userLabel(id) {
  return meta.users.find((u) => u.id === id)?.name || '鈥?
}

function canCreate() {
  return auth.role === 'admin' || auth.role === 'reviewer'
}

defineExpose({ openCreate: () => { createVisible.value = true } })

function mapTagColor(t) {
  const m = { danger: 'error', warning: 'warning', success: 'success', info: 'default', primary: 'processing', '': 'default' }
  return m[t] || t || 'default'
}

const ornTableCols1 = [
  { title: '宸ュ崟鍙?, key: 'id', dataIndex: 'id', width: 160, fixed: 'left' },
  { title: '鐘舵€?, key: '鐘舵€?, width: 100 },
  { title: '浼樺厛绾?, key: '浼樺厛绾?, width: 80 },
  { title: '澶勭悊浜?, key: '澶勭悊浜?, width: 100 },
  { title: '鍗忎綔', key: '鍗忎綔', width: 80 },
  { title: '杩涘害', key: '杩涘害', width: 110 },
  { title: '鏇存柊鏃堕棿', key: 'updatedAt', dataIndex: 'updatedAt', width: 160 }
]
</script>

<template>
  <div>
    <div class="list-actions">
      <a-button v-if="canCreate()" type="primary" @click="createVisible = true">Create job</a-button>
    </div>
    <a-card  class="filter-card">
      <a-form  @submit.prevent>
        <a-form-item label="鍏抽敭瀛?>
          <a-input v-model:value="filters.keyword" allow-clear placeholder="宸ュ崟鍙?鏍囬" style="width: 180px" />
        </a-form-item>
        <a-form-item label="鐘舵€?>
          <a-select v-model:value="filters.status" allow-clear placeholder="鍏ㄩ儴" style="width: 130px">
            <a-select-option v-for="s in meta.statuses" :key="s.value" :value="s.value">{{ s.label }}</a-select-option>
          </a-select>
        </a-form-item>
        <a-form-item label="妯℃€?>
          <a-select v-model:value="filters.modality" allow-clear placeholder="鍏ㄩ儴" style="width: 150px">
            <a-select-option v-for="m in meta.modalities" :key="m.value" :value="m.value">{{ m.label }}</a-select-option>
          </a-select>
        </a-form-item>
        <a-form-item>
          <a-button type="primary" :loading="loading" @click="onSearch">Search</a-button>
        </a-form-item>
      </a-form>
    </a-card>

    <a-card>
      <a-empty v-if="!tasks?.length && !loading" description="No jobs yet 鈥?Create job" />
    <a-table :data-source="tasks" :columns="ornTableCols1" :pagination="false" row-key="id" :loading="loading" v-show="tasks?.length">
  <template #bodyCell="{ column, record }">
      <template v-if="column.key === 'id'">
{{ getModalityMeta(record.modality)?.short }}
      </template>
      <template v-if="column.key === '鐘舵€?">
<a-tag size="small" :color="mapTagColor(getStatusMeta(record.status)?.color || 'info')">
              {{ getStatusMeta(record.status)?.label }}
            </a-tag>
      </template>
      <template v-if="column.key === '浼樺厛绾?">
<a-tag
              size="small"
              :color="mapTagColor(record.priority === 'high' ? 'danger' : record.priority === 'medium' ? 'warning' : 'info')">
              {{ record.priority === 'high' ? '楂? : record.priority === 'medium' ? '涓? : '浣? }}
            </a-tag>
      </template>
      <template v-if="column.key === '澶勭悊浜?">
{{ userLabel(record.assigneeId) }}
      </template>
      <template v-if="column.key === '鍗忎綔'">
{{ record.collaborators?.length || 0 }}浜?      </template>
      <template v-if="column.key === '杩涘害'">
{{ record.doneCount }}/{{ record.sampleCount }}
      </template>
      <template v-if="column.key === 'updatedAt'">
<a-button  type="" @click="goAnnotate(record.id)">鏍囨敞</a-button>
            <a-button  type="" @click="openAssign(row)">鍒嗛厤</a-button>
            <a-button  type="" @click="goAudit(record.id)">瀹℃牳</a-button>
      </template>
  </template>
</a-table>
    </a-card>

    <a-modal v-model:open="createVisible" title="鏂板缓璐ㄦ帶宸ュ崟" width="480px">
      <a-form :label-col="{ style: { width: '90px' } }">
        <a-form-item label="鏍囬" required>
          <a-input v-model:value="createForm.title" placeholder="渚嬪锛氶棬鍘呭紑闂ㄧ偣浜戝垎鍓茶繑淇? />
        </a-form-item>
        <a-form-item label="妯℃€? required>
          <a-select v-model:value="createForm.modality" style="width: 100%">
            <a-select-option v-for="m in meta.modalities" :key="m.value" :value="m.value">{{ m.label }}</a-select-option>
          </a-select>
        </a-form-item>
        <a-form-item label="浼樺厛绾?>
          <a-radio-group v-model:value="createForm.priority">
            <a-radio value="high">楂?/a-radio>
            <a-radio value="medium">涓?/a-radio>
            <a-radio value="low">浣?/a-radio>
          </a-radio-group>
        </a-form-item>
        <a-form-item label="鏍锋湰鏁?>
          <a-input-number v-model:value="createForm.sampleCount" :min="1" :max="500" />
        </a-form-item>
        <a-form-item label="鎵规鍙?>
          <a-input v-model:value="createForm.batchId" placeholder="鍙€? />
        </a-form-item>
      </a-form>
      <template #footer>
        <a-button @click="createVisible = false">鍙栨秷</a-button>
        <a-button type="primary" @click="onCreate">鍒涘缓</a-button>
      </template>
    </a-modal>

    <a-modal v-model:open="assignVisible" title="浜哄憳鍒嗛厤" width="420px">
      <p class="hint">宸ュ崟 {{ assignTarget?.id }} 路 {{ assignTarget?.title }}</p>
      <a-select v-model:value="assignUserId" placeholder="閫夋嫨鏍囨敞鍛? style="width: 100%">
        <a-select-option
          v-for="u in annotators"
          :key="u.id"
          :value="u.id">{{ `${u.name}锛?{u.dept}锛塦 }}</a-select-option>
      </a-select>
      <template #footer>
        <a-button @click="assignVisible = false">鍙栨秷</a-button>
        <a-button type="primary" @click="confirmAssign">纭鍒嗛厤</a-button>
      </template>
    </a-modal>
  </div>
</template>
